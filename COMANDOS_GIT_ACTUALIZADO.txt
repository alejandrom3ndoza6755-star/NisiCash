# ===============================================
# COMANDOS GIT PARA NISICASH
# ===============================================

## PROBLEMA: Tu rama local se llama "main" pero la remota se llama "master"

## SOLUCIÓN RÁPIDA (Usar siempre):

# Método 1: Push directo (RECOMENDADO)
git push origin HEAD:master

# Método 2: Usar el script PowerShell
.\push.ps1 "tu mensaje de commit aquí"

# ===============================================
# FLUJO DE TRABAJO COMPLETO
# ===============================================

# 1. Ver archivos modificados
git status

# 2. Agregar todos los cambios
git add .

# 3. Hacer commit con mensaje descriptivo
git commit -m "descripción de los cambios"

# 4. Subir a GitHub
git push origin HEAD:master

# ===============================================
# COMANDOS ÚTILES
# ===============================================

# Ver historial de commits
git log --oneline

# Ver diferencias antes de commit
git diff

# Deshacer cambios en un archivo (ANTES de commit)
git checkout -- nombre-archivo.js

# Ver estado actual
git status

# Crear nueva rama
git checkout -b nueva-rama

# Cambiar de rama
git checkout nombre-rama

# ===============================================
# MENSAJES DE COMMIT (Convenciones)
# ===============================================

# feat: Nueva funcionalidad
git commit -m "feat: Agregar sistema de referidos"

# fix: Corrección de bug
git commit -m "fix: Corregir enlace de Facebook"

# update: Actualización
git commit -m "update: Actualizar enlaces de redes sociales"

# docs: Documentación
git commit -m "docs: Agregar guía de redes sociales"

# style: Cambios de estilo/diseño
git commit -m "style: Mejorar diseño de botones"

# refactor: Refactorización de código
git commit -m "refactor: Optimizar función de compartir"

# ===============================================
# SOLUCIÓN DE PROBLEMAS COMUNES
# ===============================================

## Error: "fatal: The upstream branch does not match"
# Solución: Usar git push origin HEAD:master

## Error: "Changes not staged for commit"
# Solución: git add . antes de hacer commit

## Error: "Please tell me who you are"
# Solución:
git config --global user.email "alejandrom3ndoza6755@gmail.com"
git config --global user.name "Alejandro Mendoza"

## Warning: "LF will be replaced by CRLF"
# Esto es normal en Windows, puedes ignorarlo

## Cancelar un commit (antes de push)
git reset --soft HEAD~1

## Ver cambios subidos a GitHub
# Ir a: https://github.com/alejandrom3ndoza6755-star/NisiCash

# ===============================================
# RESUMEN RÁPIDO
# ===============================================

git add .
git commit -m "mensaje"
git push origin HEAD:master

# O usar el script:
.\push.ps1 "mensaje"
